\def\thischapterAppendix{%
  \putwordAppendix{} \thischapternum\space\thischaptername}

\def\thischapterChapter{%
  第\ \thischapternum \putwordChapter{}\space\thischaptername}

\def\thissectionDef{%
  \putwordAppendix{} \thissectionnum\putwordSection{}\space\thissectionname}

\def\putpageref#1{ \refx{#1-pg}\tie\putwordpage}

\def\Ynumbered{%
  \ifnum\secno=0
    第\ \the\chapno\tie\putwordChapter
  \else \ifnum\subsecno=0
    \the\chapno.\the\secno\putwordSection
  \else \ifnum\subsubsecno=0
    \the\chapno.\the\secno.\the\subsecno\tie\putwordSection
  \else
    \the\chapno.\the\secno.\the\subsecno.\the\subsubsecno\tie\putwordSection
  \fi\fi\fi
}

      \let\zhfont#1\let\zhpunctfont#1%

    % ------------------------------------------------------------------
    % Western font override (project asymptote-zh).
    % Target: same effect as LaTeX \usepackage[tt=false,osf]{libertinus},
    % but with mono = Latin Modern Mono (not Libertinus Mono).
    %   rm/it/bf/sc = Libertinus Serif (+onum oldstyle figures, +smcp for sc)
    %   sf          = Libertinus Sans
    %   tt/ttsl/ttb = Latin Modern Mono (regular/italic/bold)
    % Math fonts (\texti=cmmi, \textsy=cmsy) stay Computer Modern.
    % All sizes flow through \setfont, so one override covers every
    % size group (text/small/smaller/seven/title/chap/sec/defun...).
    % Files are shipped with TeX Live (libertinus-fonts, lm) and are
    % found by filename via kpathsea -> portable across machines.
    % ------------------------------------------------------------------
    \gdef\setfont#1#2#3#4#5{%
      \font#1=\westfontfile{#2} at \dimexpr #3pt/1000*#4\relax
      \csname cmap#5\endcsname#1%
      \ifx#2\ttshape\hyphenchar#1=-1 \fi
      \ifx#2\ttbshape\hyphenchar#1=-1 \fi
      \ifx#2\ttslshape\hyphenchar#1=-1 \fi
    }
    \def\westfontfile#1{%
      \ifx#1\rmshape  "[LibertinusSerif-Regular.otf]:+onum;mapping=tex-text"%
      \else\ifx#1\bfshape "[LibertinusSerif-Bold.otf]:+onum;mapping=tex-text"%
      \else\ifx#1\itshape "[LibertinusSerif-Italic.otf]:+onum;mapping=tex-text"%
      \else\ifx#1\slshape "[LibertinusSerif-Italic.otf]:+onum;mapping=tex-text"%
      \else\ifx#1\scshape "[LibertinusSerif-Regular.otf]:+smcp;+onum;mapping=tex-text"%
      \else\ifx#1\sfshape "[LibertinusSans-Regular.otf]:mapping=tex-text"%
      \else\ifx#1\ttshape "[lmmono10-regular.otf]:mapping=tex-text"%
      \else\ifx#1\ttbshape "[lmmonolt10-bold.otf]:mapping=tex-text"%
      \else\ifx#1\ttslshape "[lmmono10-italic.otf]:mapping=tex-text"%
      % Bold-variant shapes (used by title/chap/sec/subsec heading groups,
      % where the normal face is bold). Without these the heading Latin
      % fell back to cmr10.
      \else\ifx#1\rmbshape "[LibertinusSerif-Bold.otf]:+onum;mapping=tex-text"%
      \else\ifx#1\itbshape "[LibertinusSerif-BoldItalic.otf]:+onum;mapping=tex-text"%
      \else\ifx#1\slbshape "[LibertinusSerif-BoldItalic.otf]:+onum;mapping=tex-text"%
      \else\ifx#1\sfbshape "[LibertinusSans-Bold.otf]:mapping=tex-text"%
      \else\ifx#1\scbshape "[LibertinusSerif-Bold.otf]:+onum;mapping=tex-text"%
      \else cmr10% fallback: should not happen
      \fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi
    }

    % ------------------------------------------------------------------
    % CJK/Latin boundary glue fix (project asymptote-zh).
    % zhspacing's \zhs@skipspaces eats spaces after hanzi, but the
    % original \zhs@skipsp@ces only restores glue when the next token is
    % a letter/other char or math '$'. When the next token is a group
    % opening '{' (inline @code{...}, @samp{...}) or a control sequence
    % (@math after a space, \noindent, ...), the space is gone and NO
    % glue is inserted, so hanzi collides with the following Latin with
    % zero gap. Extend the final else to always insert \skipenzh.
    % ------------------------------------------------------------------
    {\catcode64=11
    % Robust font save/restore without grouping: the original
    % \zhgroupsavefont uses \begingroup/\endgroup pairs which go
    % unbalanced when a hanzi run starts at a list boundary (no save
    % token fires) but ends at a boundary token (restore fires),
    % producing spurious "Missing }" errors. Capture and re-select
    % the font instead.
    \gdef\zhs@savefont{\edef\zhs@savedfont{\the\font}}%
    \gdef\zhs@restorefont{\zhs@savedfont}%
    \gdef\zhs@skipsp@ces{%
      \ifcat a\noexpand\zhs@tmpchar % catcode 11 (letter)
        \zhs@testskipenzh
      \else
        \ifcat !\noexpand\zhs@tmpchar % catcode 12 (other char)
          \zhs@testskipenzh
        \else
          \ifcat $\noexpand\zhs@tmpchar % catcode 3 (math shift)
            \skipenzh\relax
          \else
            \skipenzh\relax
          \fi
        \fi
      \fi
    }\catcode64=0 }

    % Interword space fix (project asymptote-zh).
